   //--- logging
   void LogError(string msg)   { if(InpLogLevel >= LOGL_ERROR)   Print("[ERROR] ", msg); }
   void LogWarning(string msg) { if(InpLogLevel >= LOGL_WARNING) Print("[WARN] ",  msg); }
   void LogInfo(string msg)    { if(InpLogLevel >= LOGL_INFO)    Print("[INFO] ",  msg); }
   void LogDebug(string msg)   { if(InpLogLevel >= LOGL_DEBUG)   Print("[DEBUG] ", msg); }

   //--- helpers
   long GenerateMagicNumber()
   {
      string str = _Symbol + IntegerToString((int)Period());
      uchar arr[];
      StringToCharArray(str, arr);
      long hash = 0;
      for(int i = 0; i < ArraySize(arr); i++)
         hash = (hash * 31 + arr[i]) % 9999999;
      return 100000 + hash;
   }

   bool IsRecoverableError(uint code)
   {
      switch(code)
      {
         case TRADE_RETCODE_REQUOTE:
         case TRADE_RETCODE_PRICE_CHANGED:
         case TRADE_RETCODE_PRICE_OFF:
         case TRADE_RETCODE_TIMEOUT:
         case TRADE_RETCODE_CONNECTION:
         case TRADE_RETCODE_TOO_MANY_REQUESTS:
            return true;
      }
      return false;
   }

   //--- broker's minimum allowed distance between price and SL/TP
   double StopMinDist()
   {
      long stopLvl   = SymbolInfoInteger(_Symbol, SYMBOL_TRADE_STOPS_LEVEL);
      long freezeLvl = SymbolInfoInteger(_Symbol, SYMBOL_TRADE_FREEZE_LEVEL);
      double base = (double)(MathMax(stopLvl, freezeLvl) + 1) * _Point;
      double safe = base + StopSafetyMarginPts * _Point;
      return safe;
   }

   //--- time filters
   bool IsInKillZone()
   {
      if(TradeAllSessions) return true;
      if(!UseSilverBullet) return true;

      MqlDateTime dt;
      TimeToStruct(TimeCurrent(), dt);
      int hour = dt.hour;
      if(LondonSilverBullet && hour >= LondonKillZoneStart && hour < LondonKillZoneEnd) return true;
      if(NewYorkSilverBullet && hour >= NYKillZoneStart && hour < NYKillZoneEnd) return true;
      return false;
   }

   bool IsInTradingSession()
   {
      if(TradeAllSessions) return true;

      MqlDateTime dt;
      TimeToStruct(TimeCurrent(), dt);
      int hour = dt.hour;
      if(TradeAsianSession   && hour >= 0 && hour < 6) return true;
      if(TradeLondonSession  && hour >= LondonStartHour  && hour < LondonStartHour + 4) return true;
      if(TradeNewYorkSession && hour >= NewYorkStartHour && hour < NewYorkStartHour + 4) return true;
      return false;
   }

   bool TimeFilterOK()
   {
      if(!IsInTradingSession()) return false;
      if(UseSilverBullet && !IsInKillZone()) return false;
      return true;
   }

   //--- spread bands
   bool SpreadFilterOK(double spreadPts)
   {
      if(spreadPts < Spread_Min)  return false;
      if(spreadPts > Spread_Max)  return false;
      return true;
   }

   string SpreadStatus(double spreadPts)
   {
      if(spreadPts < Spread_Min)  return "TOO TIGHT";
      if(spreadPts > Spread_Max)  return "TOO HIGH";
      if(MathAbs(spreadPts - Spread_Normal) <= 5.0) return "NORMAL";
      return "OK";
   }

   //--- bias
   int GetMarketBias()
   {
      double ema[];
      ArraySetAsSeries(ema, true);
      if(CopyBuffer(m_emaHandle, 0, 0, 1, ema) <= 0) return 0;
      double price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      if(price > ema[0]) return  1;
      if(price < ema[0]) return -1;
      return 0;
   }

   //--- ICT structures (kept for dashboard, unused for signal in v4.02)
   int CalculateLevelStrength(double price)
   {
      int strength = 0;
      double tolerance = m_currentATR * 0.1;
      for(int i = 1; i < 100; i++)
      {
         double high = iHigh(_Symbol, PERIOD_M5, i);
         double low  = iLow(_Symbol,  PERIOD_M5, i);
         if(MathAbs(high - price) < tolerance || MathAbs(low - price) < tolerance)
            strength++;
      }
      return strength;
   }

   void AddLiquidityLevel(double price, bool isHigh, datetime time)
   {
      int size = ArraySize(m_liquidityLevels);
      ArrayResize(m_liquidityLevels, size + 1);
      m_liquidityLevels[size].price    = price;
      m_liquidityLevels[size].isHigh   = isHigh;
      m_liquidityLevels[size].time     = time;
      m_liquidityLevels[size].swept    = false;
      m_liquidityLevels[size].volume   = 0.0;
      m_liquidityLevels[size].strength = CalculateLevelStrength(price);
   }

   void UpdateLiquidityLevels()
   {
      ArrayResize(m_liquidityLevels, 0);
      if(iBars(_Symbol, PERIOD_M5) < LiquidityLookback + 120) return;

      for(int i = 3; i < LiquidityLookback; i++)
      {
         double h = iHigh(_Symbol, PERIOD_M5, i);
         double l = iLow(_Symbol,  PERIOD_M5, i);

         if(h > iHigh(_Symbol, PERIOD_M5, i - 1) && h > iHigh(_Symbol, PERIOD_M5, i - 2) &&
            h > iHigh(_Symbol, PERIOD_M5, i + 1) && h > iHigh(_Symbol, PERIOD_M5, i + 2))
            AddLiquidityLevel(h, true, iTime(_Symbol, PERIOD_M5, i));

         if(l < iLow(_Symbol, PERIOD_M5, i - 1) && l < iLow(_Symbol, PERIOD_M5, i - 2) &&
            l < iLow(_Symbol, PERIOD_M5, i + 1) && l < iLow(_Symbol, PERIOD_M5, i + 2))
            AddLiquidityLevel(l, false, iTime(_Symbol, PERIOD_M5, i));
      }

      AddLiquidityLevel(iHigh(_Symbol, PERIOD_D1, 1), true,  iTime(_Symbol, PERIOD_D1, 1));
      AddLiquidityLevel(iLow(_Symbol,  PERIOD_D1, 1), false, iTime(_Symbol, PERIOD_D1, 1));
      AddLiquidityLevel(iHigh(_Symbol, PERIOD_W1, 1), true,  iTime(_Symbol, PERIOD_W1, 1));
      AddLiquidityLevel(iLow(_Symbol,  PERIOD_W1, 1), false, iTime(_Symbol, PERIOD_W1, 1));
   }

   void UpdateFairValueGaps()
   {
      ArrayResize(m_fairValueGaps, 0);
      if(!UseFairValueGaps) return;
      if(iBars(_Symbol, PERIOD_M5) < FVG_Lookback + 5) return;

      for(int i = 2; i < FVG_Lookback; i++)
      {
         int a = i + 1;
         int c = i - 1;
         double highA = iHigh(_Symbol, PERIOD_M5, a), lowA = iLow(_Symbol, PERIOD_M5, a);
         double highC = iHigh(_Symbol, PERIOD_M5, c), lowC = iLow(_Symbol, PERIOD_M5, c);

         if(lowC > highA)
         {
            double gap = lowC - highA;
            if(gap >= m_currentATR * FVG_MinSizeATR)
            {
               int n = ArraySize(m_fairValueGaps);
               ArrayResize(m_fairValueGaps, n + 1);
               m_fairValueGaps[n].topPrice    = lowC;
               m_fairValueGaps[n].bottomPrice = highA;
               m_fairValueGaps[n].time        = iTime(_Symbol, PERIOD_M5, i);
               m_fairValueGaps[n].isBullish   = true;
               m_fairValueGaps[n].filled      = false;
               m_fairValueGaps[n].size        = gap;
            }
         }
         if(highC < lowA)
         {
            double gap = lowA - highC;
            if(gap >= m_currentATR * FVG_MinSizeATR)
            {
               int n = ArraySize(m_fairValueGaps);
               ArrayResize(m_fairValueGaps, n + 1);
               m_fairValueGaps[n].topPrice    = lowA;
               m_fairValueGaps[n].bottomPrice = highC;
               m_fairValueGaps[n].time        = iTime(_Symbol, PERIOD_M5, i);
               m_fairValueGaps[n].isBullish   = false;
               m_fairValueGaps[n].filled      = false;
               m_fairValueGaps[n].size        = gap;
            }
         }
      }
   }

   void UpdateOrderBlocks()
   {
      ArrayResize(m_orderBlocks, 0);
      if(!UseOrderBlocks) return;
      if(iBars(_Symbol, PERIOD_M5) < 60) return;

      for(int i = 2; i < 50; i++)
      {
         double oi = iOpen(_Symbol,  PERIOD_M5, i), ci = iClose(_Symbol, PERIOD_M5, i);
         double hi = iHigh(_Symbol,  PERIOD_M5, i), li = iLow(_Symbol,   PERIOD_M5, i);
         double on = iOpen(_Symbol,  PERIOD_M5, i - 1), cn = iClose(_Symbol, PERIOD_M5, i - 1);

         if(ci < oi && cn > on && cn > hi)
         {
            int n = ArraySize(m_orderBlocks);
            ArrayResize(m_orderBlocks, n + 1);
            m_orderBlocks[n].topPrice    = hi;
            m_orderBlocks[n].bottomPrice = li;
            m_orderBlocks[n].time        = iTime(_Symbol, PERIOD_M5, i);
            m_orderBlocks[n].isBullish   = true;
            m_orderBlocks[n].mitigated   = false;
            m_orderBlocks[n].volume      = (double)iVolume(_Symbol, PERIOD_M5, i);
         }
         if(ci > oi && cn < on && cn < li)
         {
            int n = ArraySize(m_orderBlocks);
            ArrayResize(m_orderBlocks, n + 1);
            m_orderBlocks[n].topPrice    = hi;
            m_orderBlocks[n].bottomPrice = li;
            m_orderBlocks[n].time        = iTime(_Symbol, PERIOD_M5, i);
            m_orderBlocks[n].isBullish   = false;
            m_orderBlocks[n].mitigated   = false;
            m_orderBlocks[n].volume      = (double)iVolume(_Symbol, PERIOD_M5, i);
         }
      }
   }

   void UpdateATR()
   {
      double atrBuffer[];
      ArraySetAsSeries(atrBuffer, true);
      if(CopyBuffer(m_atrHandle, 0, 0, 2, atrBuffer) >= 2)
         m_currentATR = atrBuffer[1];
   }

   //--- signal: v4.02 is bias-only, confluence is dead code (kept for future)
   int GetSignal()
   {
      int bias = GetMarketBias();
      if(bias == 0) return 0;
      return bias;    // bias-only, always
   }

   //--- daily stats
   void ResetDailyStats()
   {
      m_todayTradeCount = 0;
      m_dayStartBalance = m_account.Balance();
   }

   void CheckNewDay()
   {
      datetime d = iTime(_Symbol, PERIOD_D1, 0);
      if(d != m_dayStart)
      {
         m_dayStart = d;
         ResetDailyStats();
      }
   }

   bool CheckDailyLimits()
   {
      if(MaxDailyTrades > 0 && m_todayTradeCount >= MaxDailyTrades) return true;

      if(m_dayStartBalance > 0)
      {
         double dailyPL = (m_account.Equity() - m_dayStartBalance) / m_dayStartBalance * 100.0;
         if(MaxDailyRiskPercent   > 0 && dailyPL <= -MaxDailyRiskPercent)   return true;
         if(MaxDailyProfitPercent > 0 && dailyPL >=  MaxDailyProfitPercent) return true;
      }
      return false;
   }

   void CountMyPositions(int &buys, int &sells)
   {
      buys = 0; sells = 0;
      for(int i = PositionsTotal() - 1; i >= 0; i--)
      {
         if(!m_position.SelectByIndex(i)) continue;
         if(m_position.Symbol() != _Symbol || m_position.Magic() != m_magicNumber) continue;
         if(m_position.PositionType() == POSITION_TYPE_BUY) buys++; else sells++;
      }
   }

   bool CanOpenNewPosition(bool isBuy)
   {
      int buys, sells;
      CountMyPositions(buys, sells);

      if(MaxPositions > 0 && (buys + sells) >= MaxPositions) return false;
      if(!AllowOppositePositions && ((isBuy && sells > 0) || (!isBuy && buys > 0))) return false;

      double spread = (double)SymbolInfoInteger(_Symbol, SYMBOL_SPREAD);
      if(!SpreadFilterOK(spread)) return false;

      if(m_currentATR <= 0 || m_currentATR / _Point < MinATR_Points) return false;
      if(!TimeFilterOK()) return false;

      //--- cooldown between trades
      if(SecondsBetweenTrades > 0 &&
         (TimeCurrent() - m_lastEntryTime) < SecondsBetweenTrades) return false;

      return true;
   }

   double NormalizeVolume(double lot)
   {
      double minLot  = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
      double maxLot  = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
      double stepLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
      if(stepLot <= 0) stepLot = 0.01;
      if(minLot  <= 0) minLot  = 0.01;
      if(maxLot  <= 0) maxLot  = 100.0;

      lot = MathFloor(lot / stepLot + 1e-9) * stepLot;
      lot = MathMax(minLot, MathMin(maxLot, lot));
      int digits = (int)MathMax(0, MathCeil(-MathLog10(stepLot)));
      return NormalizeDouble(lot, digits);
   }

   //--- $30 -> 0.01 lot scaling
   double CalcBalanceLot()
   {
      double balance = m_account.Balance();
      if(balance < MoneyPerMicroLot) return NormalizeVolume(FixedLotFallback);
      double microUnits = MathFloor(balance / MoneyPerMicroLot);
      double lot = microUnits * 0.01;
      if(lot > MaxLotCap) lot = MaxLotCap;
      return NormalizeVolume(lot);
   }

   //--- v4.02: broker-safe levels. SL and TP are widened if needed to satisfy stops level.
   void BuildLevels(bool isBuy, double &entry, double &sl, double &tp)
   {
      double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      double minDist = StopMinDist();                     // >= broker stops level + margin

      double slDist = MathMax(SL_Points * _Point, minDist);
      double tpDist = MathMax(TP_Points * _Point, minDist);

      if(isBuy)
      {
         entry = ask;
         sl = entry - slDist;
         tp = entry + tpDist;
         // clamp SL below bid - minDist
         if(sl > bid - minDist) sl = bid - minDist;
      }
      else
      {
         entry = bid;
         sl = entry + slDist;
         tp = entry - tpDist;
         if(sl < ask + minDist) sl = ask + minDist;
      }

      entry = NormalizeDouble(entry, _Digits);
      sl    = NormalizeDouble(sl,    _Digits);
      tp    = NormalizeDouble(tp,    _Digits);

      // final validation: SL/TP must be on the correct side of current price
      if(isBuy)
      {
         if(sl >= bid - _Point) sl = NormalizeDouble(bid - minDist, _Digits);
         if(tp <= ask + _Point) tp = NormalizeDouble(ask + minDist, _Digits);
      }
      else
      {
         if(sl <= ask + _Point) sl = NormalizeDouble(ask + minDist, _Digits);
         if(tp >= bid - _Point) tp = NormalizeDouble(bid - minDist, _Digits);
      }
   }

   void ExecuteTrade(bool isBuy)
   {
      if(!CanOpenNewPosition(isBuy) || CheckDailyLimits()) return;

      double entry, sl, tp;
      BuildLevels(isBuy, entry, sl, tp);
      double lot = CalcBalanceLot();

      //--- margin safety
      double margin = 0;
      ENUM_ORDER_TYPE ot = isBuy ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
      double stepLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
      double minLot  = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
      if(stepLot <= 0) stepLot = 0.01;
      if(minLot  <= 0) minLot  = 0.01;

      while(lot > minLot && OrderCalcMargin(ot, _Symbol, lot, entry, margin) &&
            margin > m_account.FreeMargin() * 0.8)
         lot = NormalizeVolume(lot - stepLot);

      bool ok = false;
      uint rc = 0;
      for(int attempt = 0; attempt < OrderRetryCount; attempt++)
      {
         m_symbol.RefreshRates();
         BuildLevels(isBuy, entry, sl, tp);

         bool sent = isBuy ? m_trade.Buy(lot,  _Symbol, 0, sl, tp, "BAKOME_ICT_EA")
                           : m_trade.Sell(lot, _Symbol, 0, sl, tp, "BAKOME_ICT_EA");
         rc = m_trade.ResultRetcode();

         if(sent && (rc == TRADE_RETCODE_DONE ||
                     rc == TRADE_RETCODE_PLACED ||
                     rc == TRADE_RETCODE_DONE_PARTIAL))
         {
            ok = true;
            break;
         }
         if(!IsRecoverableError(rc)) break;
         Sleep(OrderRetryDelayMs * (1 << attempt));
      }

      if(ok)
      {
         LogInfo(StringFormat("[OPENED] %s %.2f @ %.*f  SL=%.*f  TP=%.*f  (spread=%d pts, ATR=%.0f pts)",
                              isBuy ? "BUY" : "SELL", lot,
                              _Digits, entry, _Digits, sl, _Digits, tp,
                              (int)SymbolInfoInteger(_Symbol, SYMBOL_SPREAD),
                              m_currentATR / _Point));
         m_todayTradeCount++;
         m_lastEntryTime = TimeCurrent();
      }
      else
      {
         LogError(StringFormat("[REJECTED] rc=%u (%s)  entry=%.*f sl=%.*f tp=%.*f lot=%.2f",
                               rc, m_trade.ResultRetcodeDescription(),
                               _Digits, entry, _Digits, sl, _Digits, tp, lot));
      }
   }

   //--- position state
   int StateIndex(ulong ticket)
   {
      int n = ArraySize(m_states);
      for(int i = 0; i < n; i++)
         if(m_states[i].ticket == ticket) return i;
      ArrayResize(m_states, n + 1);
      m_states[n].ticket      = ticket;
      m_states[n].partialDone = false;
      return n;
   }

   void CleanupStates()
   {
      for(int i = ArraySize(m_states) - 1; i >= 0; i--)
      {
         if(PositionSelectByTicket(m_states[i].ticket)) continue;
         int n = ArraySize(m_states);
         for(int k = i; k < n - 1; k++) m_states[k] = m_states[k + 1];
         ArrayResize(m_states, n - 1);
      }
   }

   //--- manage: SL/TP recovery, partial, breakeven, trailing
   void ManagePositions()
   {
      if(m_currentATR <= 0) return;
      CleanupStates();

      double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      if(bid <= 0 || ask <= 0) return;
      double minDist = StopMinDist();

      for(int i = PositionsTotal() - 1; i >= 0; i--)
      {
         if(!m_position.SelectByIndex(i)) continue;
         if(m_position.Symbol() != _Symbol || m_position.Magic() != m_magicNumber) continue;

         ulong  ticket = m_position.Ticket();
         bool   isBuy  = (m_position.PositionType() == POSITION_TYPE_BUY);
         double open   = m_position.PriceOpen();
         double sl     = m_position.StopLoss();
         double tp     = m_position.TakeProfit();
         double vol    = m_position.Volume();
         double price  = isBuy ? bid : ask;
         double profitDist = isBuy ? (price - open) : (open - price);

         //--- 1) recover missing SL/TP anchored to open price
         if(sl == 0.0 || tp == 0.0)
         {
            double nsl, ntp;
            double slDist = MathMax(SL_Points * _Point, minDist);
            double tpDist = MathMax(TP_Points * _Point, minDist);
            if(isBuy)
            {
               nsl = open - slDist;
               ntp = open + tpDist;
               if(nsl > bid - minDist) nsl = bid - minDist;
            }
            else
            {
               nsl = open + slDist;
               ntp = open - tpDist;
               if(nsl < ask + minDist) nsl = ask + minDist;
            }
            nsl = NormalizeDouble(nsl, _Digits);
            ntp = NormalizeDouble(ntp, _Digits);
            if(!m_trade.PositionModify(ticket, nsl, ntp))
               LogDebug(StringFormat("Recover SL/TP failed T=%I64u rc=%u", ticket, m_trade.ResultRetcode()));
            continue;
         }

         //--- 2) partial close
         if(UsePartialClose && profitDist >= m_currentATR * PartialCloseTriggerATR)
         {
            int si = StateIndex(ticket);
            if(!m_states[si].partialDone)
            {
               m_states[si].partialDone = true;
               double minLot  = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
               double stepLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
               if(stepLot <= 0) stepLot = 0.01;
               double cv = MathFloor(vol * PartialClosePercent / 100.0 / stepLot + 1e-9) * stepLot;
               if(cv >= minLot && (vol - cv) >= minLot)
               {
                  if(m_trade.PositionClosePartial(ticket, NormalizeDouble(cv, 2)))
                     continue;
               }
            }
         }

         //--- 3) breakeven + trailing
         double newSL = sl;

         if(UseBreakEven && profitDist >= m_currentATR * BE_TriggerATR)
         {
            double be = open;
            if(isBuy ? (be > newSL) : (be < newSL)) newSL = be;
         }

         if(UseTrailingStop && profitDist >= m_currentATR * Trail_StartATR)
         {
            double trail = isBuy ? price - m_currentATR * Trail_StepATR
                                 : price + m_currentATR * Trail_StepATR;
            if(isBuy ? (trail > newSL) : (trail < newSL)) newSL = trail;
         }

         newSL = isBuy ? MathMin(newSL, bid - minDist) : MathMax(newSL, ask + minDist);
         newSL = NormalizeDouble(newSL, _Digits);

         double minImprove = MathMax(_Point * 2.0, m_currentATR * 0.05);
         bool improved = isBuy ? (newSL > sl + minImprove) : (newSL < sl - minImprove);
         if(improved)
         {
            if(!m_trade.PositionModify(ticket, newSL, tp))
               LogDebug(StringFormat("Modify SL failed T=%I64u rc=%u (%s)",
                                     ticket, m_trade.ResultRetcode(),
                                     m_trade.ResultRetcodeDescription()));
         }
      }
   }

   {

         ObjectSetInteger(0, name, OBJPROP_BACK,       false);
      }

      ObjectSetInteger(0, name, OBJPROP_FONTSIZE, DashFontSize);
      ObjectSetString (0, name, OBJPROP_FONT,     "Consolas");
      ObjectSetString (0, name, OBJPROP_TEXT,     text);
   }

   void DeleteDashboard()
   {
      ObjectsDeleteAll(0, "BAKOME_DASH_");
   }

   void UpdateDashboard()
   {
      if(!ShowDashboard) { DeleteDashboard(); return; }

      int lineH   = DashFontSize + 8;
      int padX    = 10;
      int padY    = 8;
      int panelW  = 430;
      int panelH  = 32 * lineH + padY * 2;

      SetPanel("BAKOME_DASH_BG", DashX, DashY, panelW, panelH, DashBgColor);

      int x = DashX + padX;
      int y = DashY + padY;
      int line = 0;

      //---- header
      SetLabel("BAKOME_DASH_L0",
               "BAKOME ICT Gold Scalper  v4.02",
               x, y + line*lineH, DashTextColor); line++;
      SetLabel("BAKOME_DASH_L1",
               "Symbol: " + _Symbol + "   TF: M5   Bias-only mode",
               x, y + line*lineH, DashTextColor); line++;
      SetLabel("BAKOME_DASH_L2",
               "Server: " + TimeToString(TimeCurrent(), TIME_DATE|TIME_SECONDS),
               x, y + line*lineH, DashTextColor); line++;
      line++;

      //---- state
      MqlDateTime dt; TimeToStruct(TimeCurrent(), dt);
      int hour = dt.hour;

      bool inSession = IsInTradingSession();
      bool inKZ      = IsInKillZone();

      double spread  = (double)SymbolInfoInteger(_Symbol, SYMBOL_SPREAD);
      bool spreadOK  = SpreadFilterOK(spread);
      string spreadStr = SpreadStatus(spread);

      bool atrOK = (m_currentATR > 0 && m_currentATR / _Point >= MinATR_Points);

      int buys, sells;
      CountMyPositions(buys, sells);
      int totalPos = buys + sells;
      bool posOK   = (MaxPositions == 0) || (totalPos < MaxPositions);
      bool oppositeOK = AllowOppositePositions || (buys == 0 || sells == 0);

      bool dailyTradeOK = (MaxDailyTrades == 0) || (m_todayTradeCount < MaxDailyTrades);

      double dailyPL = (m_dayStartBalance > 0)
                     ? (m_account.Equity() - m_dayStartBalance) / m_dayStartBalance * 100.0
                     : 0.0;
      bool dailyLossOK   = (MaxDailyRiskPercent   <= 0) || (dailyPL > -MaxDailyRiskPercent);
      bool dailyProfitOK = (MaxDailyProfitPercent <= 0) || (dailyPL <  MaxDailyProfitPercent);
      bool dailyOK       = dailyTradeOK && dailyLossOK && dailyProfitOK;

      int bias = GetMarketBias();
      string biasStr = (bias ==  1) ? "BULLISH (price > H4 EMA200)"
                     : (bias == -1) ? "BEARISH (price < H4 EMA200)"
                                    : "NEUTRAL";

      int signal = GetSignal();
      string signalStr = (signal ==  1) ? "BUY"
                       : (signal == -1) ? "SELL"
                                        : "NONE";

      double balLot = CalcBalanceLot();

      int cooldown = SecondsBetweenTrades > 0
                   ? (int)(TimeCurrent() - m_lastEntryTime)
                   : SecondsBetweenTrades;
      bool cooldownOK = (SecondsBetweenTrades == 0) || (cooldown >= SecondsBetweenTrades);

      //---- filters
      SetLabel("BAKOME_DASH_L4", "-- FILTERS --", x, y + line*lineH, DashWarnColor); line++;

      SetLabel("BAKOME_DASH_L5",
               StringFormat("Session      : %s  (hour %02d)",
                            TradeAllSessions ? "24/5 FORCED" : (inSession ? "OPEN" : "CLOSED"),
                            hour),
               x, y + line*lineH,
               (TradeAllSessions || inSession) ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L6",
               StringFormat("Kill Zone    : %s",
                            TradeAllSessions ? "BYPASSED" :
                            (!UseSilverBullet ? "SB OFF" : (inKZ ? "OPEN" : "CLOSED"))),
               x, y + line*lineH,
               (TradeAllSessions || !UseSilverBullet || inKZ) ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L7",
               StringFormat("Spread       : %.0f pts  [min %.0f | norm %.0f | max %.0f]  %s",
                            spread, Spread_Min, Spread_Normal, Spread_Max, spreadStr),
               x, y + line*lineH, spreadOK ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L8",
               StringFormat("ATR (M5)     : %.0f / %.0f pts",
                            m_currentATR / _Point, MinATR_Points),
               x, y + line*lineH, atrOK ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L9",
               StringFormat("Positions    : %d  %s  (B%d S%d)",
                            totalPos,
                            (MaxPositions == 0 ? "[UNLIMITED]" : StringFormat("/ %d", MaxPositions)),
                            buys, sells),
               x, y + line*lineH, posOK ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L10",
               StringFormat("Trades today : %d  %s",
                            m_todayTradeCount,
                            (MaxDailyTrades == 0 ? "[UNLIMITED]" : StringFormat("/ %d", MaxDailyTrades))),
               x, y + line*lineH, dailyTradeOK ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L11",
               StringFormat("Daily P/L    : %+.2f%%  %s",
                            dailyPL,
                            (MaxDailyRiskPercent <= 0 && MaxDailyProfitPercent <= 0)
                              ? "[NO LIMITS]" : "limited"),
               x, y + line*lineH,
               (dailyLossOK && dailyProfitOK) ? DashOkColor : DashBadColor); line++;

      SetLabel("BAKOME_DASH_L12",
               StringFormat("Next lot     : %.2f  (bal $%.2f / $%.2f)",
                            balLot, m_account.Balance(), MoneyPerMicroLot),
               x, y + line*lineH, DashTextColor); line++;

      SetLabel("BAKOME_DASH_L13",
               StringFormat("Cooldown     : %s  (%ds / %ds)",
                            cooldownOK ? "READY" : "WAIT",
                            cooldown, SecondsBetweenTrades),
               x, y + line*lineH, cooldownOK ? DashOkColor : DashWarnColor); line++;

      line++;

      //---- signal
      SetLabel("BAKOME_DASH_L15", "-- ICT SIGNAL --", x, y + line*lineH, DashWarnColor); line++;

      SetLabel("BAKOME_DASH_L16",
               "Bias         : " + biasStr,
               x, y + line*lineH,
               (bias == 0) ? DashBadColor : DashOkColor); line++;

      SetLabel("BAKOME_DASH_L17",
               StringFormat("Liquidity Lv : %d  (sweeps %s)",
                            ArraySize(m_liquidityLevels),
                            UseLiquiditySweeps ? "ON" : "OFF"),
               x, y + line*lineH, DashTextColor); line++;

      SetLabel("BAKOME_DASH_L18",
               StringFormat("FVGs         : %d  (%s)",
                            ArraySize(m_fairValueGaps),
                            UseFairValueGaps ? "ON" : "OFF"),
               x, y + line*lineH, DashTextColor); line++;

      SetLabel("BAKOME_DASH_L19",
               StringFormat("Order Blocks : %d  (%s)",
                            ArraySize(m_orderBlocks),
                            UseOrderBlocks ? "ON" : "OFF"),
               x, y + line*lineH, DashTextColor); line++;

      SetLabel("BAKOME_DASH_L20",
               "Final signal : " + signalStr,
               x, y + line*lineH,
               (signal == 0) ? DashBadColor : DashOkColor); line++;

      line++;

      //---- decision
      SetLabel("BAKOME_DASH_L22", "-- DECISION --", x, y + line*lineH, DashWarnColor); line++;

      string decision;
      color  decisionColor = DashTextColor;

      if(!inSession)
      {
         decision = "WAITING : outside trading session";
         decisionColor = DashBadColor;
      }
      else if(!inKZ)
      {
         decision = "WAITING : outside Silver Bullet kill zone";
         decisionColor = DashBadColor;
      }
      else if(spread < Spread_Min)
      {
         decision = StringFormat("BLOCKED : spread %.0f < min %.0f (too tight)", spread, Spread_Min);
         decisionColor = DashBadColor;
      }
      else if(spread > Spread_Max)
      {
         decision = StringFormat("BLOCKED : spread %.0f > max %.0f (too high)", spread, Spread_Max);
         decisionColor = DashBadColor;
      }
      else if(!atrOK)
      {
         decision = "BLOCKED : ATR too low";
         decisionColor = DashBadColor;
      }
      else if(!dailyOK)
      {
         decision = "BLOCKED : daily limits reached";
         decisionColor = DashBadColor;
      }
      else if(!posOK)
      {
         decision = "BLOCKED : max positions reached";
         decisionColor = DashBadColor;
      }
      else if(!oppositeOK)
      {
         decision = "BLOCKED : opposite position open";
         decisionColor = DashBadColor;
      }
      else if(!cooldownOK)
      {
         decision = StringFormat("WAITING : cooldown %ds/%ds", cooldown, SecondsBetweenTrades);
         decisionColor = DashWarnColor;
      }
      else if(signal != 0)
      {
         decision = (signal == 1) ? "READY : BUY signal" : "READY : SELL signal";
         decisionColor = DashOkColor;
      }
      else
      {
         decision = "WAITING : neutral bias";
         decisionColor = DashWarnColor;
      }

      SetLabel("BAKOME_DASH_L23", decision, x, y + line*lineH, decisionColor); line++;

      //---- trailing / SL-TP
      line++;
      SetLabel("BAKOME_DASH_L25", "-- TRAILING / SL-TP --", x, y + line*lineH, DashWarnColor); line++;
      SetLabel("BAKOME_DASH_L26",
               StringFormat("Fixed SL/TP  : %.0f / %.0f pts   (broker min %.0f pts)",
                            SL_Points, TP_Points, StopMinDist() / _Point),
               x, y + line*lineH, DashTextColor); line++;
      SetLabel("BAKOME_DASH_L27",
               StringFormat("Trailing     : %s  (start %.2f ATR, step %.2f ATR)",
                            UseTrailingStop ? "ON" : "OFF",
                            Trail_StartATR, Trail_StepATR),
               x, y + line*lineH,
               UseTrailingStop ? DashOkColor : DashBadColor); line++;
      SetLabel("BAKOME_DASH_L28",
               StringFormat("Break-even   : %s  (at %.2f ATR)",
                            UseBreakEven ? "ON" : "OFF", BE_TriggerATR),
               x, y + line*lineH,
               UseBreakEven ? DashOkColor : DashBadColor); line++;

      //---- open positions
      line++;
      SetLabel("BAKOME_DASH_L30", "-- OPEN POSITIONS --", x, y + line*lineH, DashWarnColor); line++;

      int shown = 0;
      for(int i = PositionsTotal() - 1; i >= 0 && shown < 5; i--)
      {
         if(!m_position.SelectByIndex(i)) continue;
         if(m_position.Symbol() != _Symbol || m_position.Magic() != m_magicNumber) continue;

         ulong ticket = m_position.Ticket();
         bool  isBuy  = (m_position.PositionType() == POSITION_TYPE_BUY);
         double vol   = m_position.Volume();
         double open  = m_position.PriceOpen();
         double sl    = m_position.StopLoss();
         double tp    = m_position.TakeProfit();
         double prof  = m_position.Profit() + m_position.Swap() + m_position.Commission();

         SetLabel("BAKOME_DASH_P" + IntegerToString(shown),
                  StringFormat("%s %.2f @ %.*f  SL %.*f  TP %.*f  P/L %+.2f",
                               isBuy ? "BUY " : "SELL", vol,
                               _Digits, open, _Digits, sl, _Digits, tp, prof),
                  x, y + line*lineH,
                  (prof >= 0) ? DashOkColor : DashBadColor);
         line++; shown++;
      }
      if(shown == 0)
      {
         SetLabel("BAKOME_DASH_PNONE", "  (none)", x, y + line*lineH, DashTextColor);
         line++;
      }

      ChartRedraw(0);
   }

public:
   CUltimateICTGoldScalper()
   {
      m_atrHandle       = INVALID_HANDLE;
      m_emaHandle       = INVALID_HANDLE;
      m_dayStartBalance = 0;
      m_dayStart        = 0;
      m_todayTradeCount = 0;
      m_lastBarTime     = 0;
      m_lastEntryTime   = 0;
      m_currentATR      = 0;
      m_magicNumber     = 0;
      m_initialized     = false;
   }

   ~CUltimateICTGoldScalper() {}

   bool Init()
   {
      if(!m_symbol.Name(_Symbol))
      {
         LogError("Symbol init failed");
         return false;
      }
      m_symbol.Refresh();

      m_atrHandle = iATR(_Symbol, PERIOD_M5, 14);
      m_emaHandle = iMA(_Symbol, PERIOD_H4, 200, 0, MODE_EMA, PRICE_CLOSE);
      if(m_atrHandle == INVALID_HANDLE || m_emaHandle == INVALID_HANDLE)
      {
         LogError("Indicators failed");
         return false;
      }

      m_magicNumber = GenerateMagicNumber();
      m_trade.SetExpertMagicNumber((ulong)m_magicNumber);
      m_trade.SetDeviationInPoints(SlippagePoints);
      m_trade.SetTypeFillingBySymbol(_Symbol);
      m_trade.SetAsyncMode(false);

      m_dayStart = iTime(_Symbol, PERIOD_D1, 0);
      ResetDailyStats();
      UpdateATR();

      m_initialized = true;
      LogInfo("BAKOME EA v4.02 initialized. Magic: " + IntegerToString(m_magicNumber));
      LogInfo(StringFormat("Lot: $%.0f -> 0.01 | SL %.0f | TP %.0f | Spread %.0f-%.0f | BrokerMinStop %.0f pts | Cooldown %ds | Trailing ON",
                           MoneyPerMicroLot, SL_Points, TP_Points, Spread_Min, Spread_Max,
                           StopMinDist() / _Point, SecondsBetweenTrades));
      UpdateDashboard();
      return true;
   }

   void Deinit()
   {
      if(m_atrHandle != INVALID_HANDLE) IndicatorRelease(m_atrHandle);
      if(m_emaHandle != INVALID_HANDLE) IndicatorRelease(m_emaHandle);
      m_atrHandle = INVALID_HANDLE;
      m_emaHandle = INVALID_HANDLE;
      DeleteDashboard();
   }

   void OnTick()
   {
      if(!m_initialized) return;

      m_symbol.RefreshRates();
      UpdateATR();
      CheckNewDay();
      ManagePositions();

      datetime bar = iTime(_Symbol, PERIOD_M5, 0);
      if(bar != m_lastBarTime)
      {
         m_lastBarTime = bar;
         UpdateLiquidityLevels();
         UpdateFairValueGaps();
         UpdateOrderBlocks();
      }

      UpdateDashboard();

      //--- trade on every tick (not every bar), paced by cooldown
      int signal = GetSignal();
      if(signal ==  1) ExecuteTrade(true);
      else if(signal == -1) ExecuteTrade(false);
   }
};
