   long GenerateMagicNumber() {
      string str = _Symbol + IntegerToString(Period());
      uchar arr[];
      StringToCharArray(str, arr);
      long hash = 0;
      for(int i = 0; i < ArraySize(arr); i++)
         hash = (hash * 31 + arr[i]) % 9999999;
      return 100000 + hash;
   }

   ENUM_ORDER_TYPE_FILLING GetFillingMode() {
      return ORDER_FILLING_IOC;
   }

   bool IsRecoverableError(int errorCode) {
      switch(errorCode) {
         case TRADE_RETCODE_REQUOTE:
         case TRADE_RETCODE_PRICE_CHANGED:
         case TRADE_RETCODE_TIMEOUT:
         case TRADE_RETCODE_CONNECTION:
            return true;
         default: return false;
      }
   }

   bool ExecuteWithRetry(MqlTradeRequest &request, MqlTradeResult &result) {
      for(int attempt = 0; attempt < OrderRetryCount; attempt++) {
         ZeroMemory(result);
         if(OrderSend(request, result)) {
            if(result.retcode == TRADE_RETCODE_DONE) return true;
         }
         if(!IsRecoverableError((int)result.retcode)) break;
         Sleep(OrderRetryDelayMs * (int)MathPow(2, attempt));
      }
      return false;
   }

   string ServerTimeToPKT(datetime serverTime) {
      datetime gmt = serverTime - TimeGMTOffset();
      datetime pkt = gmt + 5 * 3600;
      return TimeToString(pkt, TIME_DATE|TIME_MINUTES|TIME_SECONDS) + " PKT";
   }

   int GetMarketBias() {
      double buf[];
      ArraySetAsSeries(buf, true);
      
      if(CopyBuffer(m_ema9Handle, 0, 0, 1, buf) > 0)   m_ema9  = buf[0];
      if(CopyBuffer(m_ema21Handle, 0, 0, 1, buf) > 0)  m_ema21 = buf[0];
      if(CopyBuffer(m_ema200Handle, 0, 0, 1, buf) > 0) m_ema200 = buf[0];

      double price = iClose(_Symbol, PERIOD_M5, 0);
      
      if(price > m_ema200 && m_ema9 > m_ema21) return POSITION_TYPE_BUY;
      if(price < m_ema200 && m_ema9 < m_ema21) return POSITION_TYPE_SELL;
      return -1;
   }

   double GetPriceDistanceInPoints(int points) {
      // Hardcoded 0.01 for Gold since the broker uses 2 decimal places
      return points * 0.01; 
   }

   void UpdateMarketData() {
      double atrBuffer[];
      ArraySetAsSeries(atrBuffer, true);
      if(CopyBuffer(m_atrHandle, 0, 0, 1, atrBuffer) > 0)
         m_currentATR = atrBuffer[0];
         
      double slDist = GetPriceDistanceInPoints(FixedSL_Points);
      double tpDist = GetPriceDistanceInPoints(FixedTP_Points);
      
      double currentAsk = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      double currentBid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      
      if(currentAsk <= 0 || currentBid <= 0) return; // Wait for valid prices
      
      m_calcEntryBuy  = currentAsk;
      m_calcBuySL     = NormalizeDouble(m_calcEntryBuy - slDist, (int)m_symbol.Digits());
      m_calcBuyTP     = NormalizeDouble(m_calcEntryBuy + tpDist, (int)m_symbol.Digits());
      
      m_calcEntrySell = currentBid;
      m_calcSellSL    = NormalizeDouble(m_calcEntrySell + slDist, (int)m_symbol.Digits());
      m_calcSellTP    = NormalizeDouble(m_calcEntrySell - tpDist, (int)m_symbol.Digits());
   }

   void CheckDailyReset() {
      MqlDateTime now, last;
      TimeToStruct(TimeCurrent(), now);
      TimeToStruct(m_lastDayReset, last);
      if(now.day != last.day || now.mon != last.mon || now.year != last.year) {
         m_todayTradeCount  = 0;
         m_dayStartBalance  = m_account.Balance();
         m_lastDayReset     = TimeCurrent();
         LogInfo("New trading day started.");
      }
   }

   bool CheckDailyLimits() {
      if(m_todayTradeCount >= MaxDailyTrades) {
         m_stopReason = "Max daily trades reached";
         m_resumeTimePKT = "Next day";
         return true;
      }
      double dailyPL = (m_account.Equity() - m_dayStartBalance) / m_dayStartBalance * 100.0;
      if(dailyPL <= -MaxDailyRiskPercent) {
         m_stopReason = "Daily loss limit reached";
         m_resumeTimePKT = "Next day";
         return true;
      }
      if(dailyPL >= MaxDailyProfitPercent) {
         m_stopReason = "Daily profit target reached";
         m_resumeTimePKT = "Next day";
         return true;
      }
      return false;
   }

   // FIXED: This function now correctly resets the "Max positions reached" status
   bool CanOpenNewPosition() {
      int openPositions = 0;
      for(int i = 0; i < PositionsTotal(); i++) {
         if(m_position.SelectByIndex(i) && m_position.Magic() == m_magicNumber && m_position.Symbol() == _Symbol)
            openPositions++;
      }
      
      // If we previously stopped due to max positions, and now we have 0, clear the stop reason
      if(openPositions == 0 && m_stopReason == "Max positions reached") {
         m_stopReason = "";
         m_resumeTimePKT = "";
      }

      if(openPositions >= MaxPositions) {
         m_stopReason = "Max positions reached";
         m_resumeTimePKT = "When position closed";
         return false;
      }
      
      double spread = (double)SymbolInfoInteger(_Symbol, SYMBOL_SPREAD);
      if(spread <= MinSpreadPoints) {
         m_stopReason = "Spread too low (" + DoubleToString(spread,1) + ") Need > 50";
         m_resumeTimePKT = "When spread > 50";
         return false;
      }
      if(spread >= MaxSpreadPoints) {
         m_stopReason = "Spread too high (" + DoubleToString(spread,1) + ") Need < 100";
         m_resumeTimePKT = "When spread < 100";
         return false;
      }
      
      if(m_currentATR / 0.01 < MinATR_Points) {
         m_stopReason = "ATR too low";
         m_resumeTimePKT = "When ATR higher";
         return false;
      }
      
      // Only clear if we aren't stopped for max positions
      if(openPositions < MaxPositions) {
         m_stopReason = "";
         m_resumeTimePKT = "";
      }
      return true;
   }

   void CalculateBullishEntry(double &entry, double &sl, double &tp) {
      entry = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      
      double slDist = GetPriceDistanceInPoints(FixedSL_Points);
      double tpDist = GetPriceDistanceInPoints(FixedTP_Points);
      
      sl = entry - slDist;
      tp = entry + tpDist;
      
      int stopsLevel = (int)SymbolInfoInteger(_Symbol, SYMBOL_TRADE_STOPS_LEVEL);
      if(stopsLevel > 0) {
         double minDist = stopsLevel * 0.01;
         if(entry - sl < minDist) sl = entry - minDist;
         if(tp - entry < minDist) tp = entry + minDist;
      }
      
      entry = NormalizeDouble(entry, (int)m_symbol.Digits());
      sl    = NormalizeDouble(sl,    (int)m_symbol.Digits());
      tp    = NormalizeDouble(tp,    (int)m_symbol.Digits());
   }

   void CalculateBearishEntry(double &entry, double &sl, double &tp) {
      entry = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      
      double slDist = GetPriceDistanceInPoints(FixedSL_Points);
      double tpDist = GetPriceDistanceInPoints(FixedTP_Points);
      
      sl = entry + slDist;
      tp = entry - tpDist;
      
      int stopsLevel = (int)SymbolInfoInteger(_Symbol, SYMBOL_TRADE_STOPS_LEVEL);
      if(stopsLevel > 0) {
         double minDist = stopsLevel * 0.01;
         if(sl - entry < minDist) sl = entry + minDist;
         if(entry - tp < minDist) tp = entry - minDist;
      }
      
      entry = NormalizeDouble(entry, (int)m_symbol.Digits());
      sl    = NormalizeDouble(sl,    (int)m_symbol.Digits());
      tp    = NormalizeDouble(tp,    (int)m_symbol.Digits());
   }

   double RiskLot(double riskPercent) {
      double riskAmount = m_account.Balance() * riskPercent / 100.0;
      double tickValue  = m_symbol.TickValue();
      double tickSize   = m_symbol.TickSize();
      if(tickSize == 0) tickSize = 0.01;
      
      double slDistance = GetPriceDistanceInPoints(FixedSL_Points);
      double lot = riskAmount / (slDistance / tickSize * tickValue);
      lot = NormalizeDouble(lot, 2);
      if(lot < 0.01) lot = 0.01;
      if(lot > 50.0) lot = 50.0;
      return lot;
   }

   void ExecuteTrade(int type) {
      if(!CanOpenNewPosition() || CheckDailyLimits()) return;
      
      double entry, sl, tp;
      if(type == ORDER_TYPE_BUY) CalculateBullishEntry(entry, sl, tp);
      else                      CalculateBearishEntry(entry, sl, tp);
      
      double lot = RiskLot(RiskPercent);

      MqlTradeRequest request = {};
      MqlTradeResult  result  = {};
      
      request.action       = TRADE_ACTION_DEAL;
      request.symbol       = _Symbol;
      request.volume       = lot;
      request.type         = (ENUM_ORDER_TYPE)type;
      request.price        = entry;
      request.sl           = sl;
      request.tp           = tp;
      request.deviation    = SlippagePoints;
      request.magic        = m_magicNumber;
      request.comment      = "BAKOME_ICT_EA";
      request.type_filling = m_filling;

      if(ExecuteWithRetry(request, result)) {
         LogInfo(StringFormat("Executed %s %.2f lots @ %.2f | SL: %.2f TP: %.2f",
                              (type==ORDER_TYPE_BUY)?"BUY":"SELL", lot, entry, sl, tp));
         m_todayTradeCount++;
         int size = ArraySize(m_activePositions);
         ArrayResize(m_activePositions, size + 1);
         m_activePositions[size].ticket     = result.order;
         m_activePositions[size].openTime   = TimeCurrent();
         m_activePositions[size].openPrice  = entry;
         m_activePositions[size].originalSL = sl;
         m_activePositions[size].originalTP = tp;
      } else {
         LogError(StringFormat("Order failed: %d | %s", result.retcode, result.comment));
      }
   }

      if(count == 0) CreateLabel("NoPos", x0, y, "No open positions", clrLabel, 9);

      ChartRedraw(0);
   }

public:
   CUltimateICTGoldScalper() {
      m_initialized = false;
      m_currentATR = 0;
      m_ema9 = m_ema21 = m_ema200 = 0;
      m_todayTradeCount = 0;
      m_dayStartBalance = 0;
      m_lastDayReset = 0;
      m_lastSignalBarTime = 0;
      m_stopReason = "";
      m_resumeTimePKT = "";
      m_calcBuySL = m_calcBuyTP = m_calcSellSL = m_calcSellTP = 0;
      m_calcEntryBuy = m_calcEntrySell = 0;
      m_filling = ORDER_FILLING_IOC;
   }

   ~CUltimateICTGoldScalper() {
      if(m_atrHandle    != INVALID_HANDLE) IndicatorRelease(m_atrHandle);
      if(m_ema9Handle   != INVALID_HANDLE) IndicatorRelease(m_ema9Handle);
      if(m_ema21Handle  != INVALID_HANDLE) IndicatorRelease(m_ema21Handle);
      if(m_ema200Handle != INVALID_HANDLE) IndicatorRelease(m_ema200Handle);
      DeleteDashboard();
   }

   void DeleteDashboard() { ObjectsDeleteAll(0, DASH_PREFIX); }

   bool Init() {
      m_symbol.Name(_Symbol);
      m_symbol.Refresh();

      m_atrHandle    = iATR(_Symbol, PERIOD_M5, 14);
      m_ema9Handle   = iMA(_Symbol, PERIOD_M5, 9,   0, MODE_EMA, PRICE_CLOSE);
      m_ema21Handle  = iMA(_Symbol, PERIOD_M5, 21,  0, MODE_EMA, PRICE_CLOSE);
      m_ema200Handle = iMA(_Symbol, PERIOD_M5, 200, 0, MODE_EMA, PRICE_CLOSE);

      if(m_atrHandle == INVALID_HANDLE || m_ema9Handle == INVALID_HANDLE || 
         m_ema21Handle == INVALID_HANDLE || m_ema200Handle == INVALID_HANDLE) {
         LogError("Failed to create indicators");
         return false;
      }

      m_filling = GetFillingMode();

      m_magicNumber     = GenerateMagicNumber();
      m_dayStartBalance = m_account.Balance();
      m_todayTradeCount = 0;
      m_lastDayReset    = TimeCurrent();
      m_initialized     = true;

      EventSetTimer(1);
      LogInfo("BAKOME EA v3.12 started | Filling: " + EnumToString(m_filling) + " | SL400 TP100");
      return true;
   }

   void OnTick() {
      if(!m_initialized) return;

      CheckDailyReset();
      UpdateMarketData();
      GetMarketBias();

      for(int i = ArraySize(m_activePositions) - 1; i >= 0; i--) {
         if(!m_position.SelectByTicket(m_activePositions[i].ticket)) {
            ArrayRemove(m_activePositions, i, 1);
            continue;
         }
      }

      // Only trade once per bar to prevent duplicate entries on every tick
      datetime currentBarTime = iTime(_Symbol, PERIOD_M5, 0);
      if(currentBarTime != m_lastSignalBarTime) {
         int bias = GetMarketBias();
         if(bias == POSITION_TYPE_BUY)  {
            ExecuteTrade(ORDER_TYPE_BUY);
            m_lastSignalBarTime = currentBarTime;
         }
         if(bias == POSITION_TYPE_SELL) {
            ExecuteTrade(ORDER_TYPE_SELL);
            m_lastSignalBarTime = currentBarTime;
         }
      }

      UpdateDashboard();
   }

   void OnTimer() {
      if(m_initialized) UpdateDashboard();
   }
};

void OnDeinit(const int reason) {

   EA.DeleteDashboard();
}

void OnTick()  { EA.OnTick(); }
